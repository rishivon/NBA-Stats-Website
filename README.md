# NBA-Stats-Website
Website for NBA stats, analysis, projections, and more

# Python Proxy
cd nba-stats-proxy
./venv/bin/python main.py   # Runs on localhost:8000

# Spring Boot Backend  
cd nba-visualizer-backend
mvn -q spring-boot:run      # Runs on localhost:8080

# Next.js Frontend
cd nba-visualizer-frontend
npm run dev                  # Runs on localhost:3000
